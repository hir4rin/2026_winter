**追加する関数**

```cpp
void BossArm::Detach()
{
	//もう上がっているならそのまま
	if (std::dynamic_pointer_cast<BossArmDetached>(m_currentState))return;

	//今の形から切り離す(戻り中なら途中の形から)
	const MATRIX now = CalcDrawSpineMatrix();
	ChangeState(std::make_shared<BossArmDetached>(GetWeakPtr()));
	m_detachSpineMatrix = now;
	m_detachMove = Vector3();
	m_detachSpinDeg = 0.0f;
	m_returnRate = 0.0f;
	m_isDetached = true;
}

void BossArm::RequestReturn()
{
	if (!m_isDetached)return;
	ChangeState(std::make_shared<BossArmReturn>(GetWeakPtr()));
}

MATRIX BossArm::CalcDetachedSpineMatrix() const
{
	//付け根を原点に戻して回してから、移動分ずらす
	const VECTOR root = VGet(m_detachSpineMatrix.m[3][0], m_detachSpineMatrix.m[3][1], m_detachSpineMatrix.m[3][2]);
	MATRIX mat = MMult(m_detachSpineMatrix, MGetTranslate(VScale(root, -1.0f)));
	mat = MMult(mat, MGetRotY(m_detachSpinDeg * kDegToRad));
	return MMult(mat, MGetTranslate(VAdd(root, m_detachMove.ToDxLibVector())));
}

MATRIX BossArm::CalcDrawSpineMatrix() const
{
	if (!m_isDetached)return CalcSpineMatrix();
	if (m_returnRate <= 0.0f)return CalcDetachedSpineMatrix();
	//戻り中//ボスは動いているので、付いた形は毎フレーム計算する
	return LerpMatrix(CalcDetachedSpineMatrix(), CalcSpineMatrix(), m_returnRate);
}

MATRIX BossArm::LerpMatrix(const MATRIX& a, const MATRIX& b, float t) const
{
	MATRIX out = MGetIdent();
	//軸ごとに向きと長さ(大きさ)を分けて補間する
	for (int i = 0; i < 3; ++i)
	{
		const VECTOR axisA = VGet(a.m[i][0], a.m[i][1], a.m[i][2]);
		const VECTOR axisB = VGet(b.m[i][0], b.m[i][1], b.m[i][2]);
		const float length = VSize(axisA) + (VSize(axisB) - VSize(axisA)) * t;
		const VECTOR axis = VScale(VNorm(VAdd(VScale(axisA, 1.0f - t), VScale(axisB, t))), length);
		out.m[i][0] = axis.x;
		out.m[i][1] = axis.y;
		out.m[i][2] = axis.z;
	}
	//位置
	for (int i = 0; i < 3; ++i)
	{
		out.m[3][i] = a.m[3][i] + (b.m[3][i] - a.m[3][i]) * t;
	}
	return out;
}
```

ポイント:
- `Detach` で **`ChangeState` を先に呼んでから値を入れる**。前のステートの Exit が値を消しても大丈夫なように。
- 行列は DxLib の `MMult(A, B)` =「A をしてから B」(`CalcSpineMatrix` と同じ書き方)。

{
	auto arm = std::dynamic_pointer_cast<BossArm>(m_owner.lock());
	if (!arm)return;

	m_timer += System::GetInstance().GetTimeScale() * arm->m_ownTimeScale;

	//上がる
	const float t = std::min(m_timer / kRiseFrame, 1.0f);
	const float rate = Easing::Apply(EasingMode::EaseOut, t, kRiseEasingPower);
	float height = kRiseHeight * rate;
	//上がりきったらふわふわ
	if (t >= 1.0f)height += sinf((m_timer - kRiseFrame) * kFloatSpeed) * kFloatWidth;
	arm->m_detachMove = Vector3(0.0f, height, 0.0f);

	//回転はだんだん止まる
	arm->m_detachSpinDeg += kSpinSpeedDeg * (1.0f - rate) * System::GetInstance().GetTimeScale() * arm->m_ownTimeScale;
}

### 1-4. BossArmReturn(新規 `Character/Enemy/BossPirate/State/BossArmReturn.h/.cpp`)

```cpp
#pragma once
#include "Character/Enemy/State/EnemyStateBase.h"
#include "Math/Vector3.h"

//切り離した腕が付け根に戻る//着いたらIdle
class BossArmReturn :
    public EnemyStateBase
{
public:
    BossArmReturn(std::weak_ptr<EnemyBase> owner);
    virtual ~BossArmReturn();

    void Enter() override;
    void Update()override;
    void Exit()override;

    const char* GetName()const override { return "BossArmReturn"; }

private:
    float m_timer = 0.0f;//経過フレーム
    Vector3 m_startMove;//戻り始めの移動
    float m_startSpinDeg = 0.0f;//戻り始めの回転
    float m_goalSpinDeg = 0.0f;//戻り終わりの回転//一番近い一周分
};
```

```cpp
#include "BossArmReturn.h"
#include "Character/Enemy/BossPirate/Part/BossArm.h"
#include "Main/System.h"
#include "Math/Easing.h"
#include <algorithm>
#include <cmath>

namespace
{
	constexpr float kReturnFrame = 30.0f;//戻るまで(F)
	constexpr float kReturnEasingPower = 2.0f;
	constexpr float kOneTurnDeg = 360.0f;
}

BossArmReturn::BossArmReturn(std::weak_ptr<EnemyBase> owner) :
	EnemyStateBase(owner)
{
}

BossArmReturn::~BossArmReturn()
{
}

void BossArmReturn::Enter()
{
	auto arm = std::dynamic_pointer_cast<BossArm>(m_owner.lock());
	if (!arm)return;
	m_timer = 0.0f;
	m_startMove = arm->m_detachMove;
	m_startSpinDeg = arm->m_detachSpinDeg;
	//回った分を戻しすぎないように、一番近い一周分へ
	m_goalSpinDeg = roundf(m_startSpinDeg / kOneTurnDeg) * kOneTurnDeg;
	arm->PlayAnim("Idle", true);
}

void BossArmReturn::Update()
{
	auto arm = std::dynamic_pointer_cast<BossArm>(m_owner.lock());
	if (!arm)return;

	m_timer += System::GetInstance().GetTimeScale() * arm->m_ownTimeScale;
	const float t = std::min(m_timer / kReturnFrame, 1.0f);
	const float rate = Easing::Apply(EasingMode::EaseInOut, t, kReturnEasingPower);

	//回転とずれを0に戻しながら、ボスに付いた形へ寄せる
	arm->m_detachMove = m_startMove * (1.0f - rate);
	arm->m_detachSpinDeg = m_startSpinDeg + (m_goalSpinDeg - m_startSpinDeg) * rate;
	arm->m_returnRate = rate;

	if (t >= 1.0f)
	{
		arm->ReturnToIdle();
		return;
	}
}

void BossArmReturn::Exit()
{
}
```

- 回転を「一番近い一周分」へ戻しながら寄せるのは、切り離した形と付いた形が真逆(180度)に近いまま `LerpMatrix` すると、
  軸が0になって形が壊れるため。回転を戻しながらなら真逆のまま補間することはない。
- `m_startMove * (1.0f - rate)` は Vector3 の `operator*(float)` がある(`Math/Vector3.h`)。
- `roundf(m_startSpinDeg / kOneTurnDeg) * kOneTurnDeg` は「今の角度に一番近い360の倍数」。
  ①何周したか(小数) → ②四捨五入して「ちょうど○周」 → ③角度に戻す。例: 170°→0°、400°→360°、560°→720°(戻るより進む方が近い)。
  360°・720° は見た目は0°と同じ向きなので、どれに戻っても背中にぴったり付き、回る量は最大でも半周で済む。

### なぜこの書き方なのか(1-3 / 1-4)
- **ステートが値を書き換え、BossArm は描くだけ**: 動きを変えるときはステートの .cpp だけ触ればよい。Step3 で吸い寄せなどを足すときも、`m_detachMove` を動かすステートを1つ足すだけ。
  ステートから private を触るために BossArm に `friend`(EnemyBase が EnemyIdle などを friend にしているのと同じ形)。
- **`dynamic_pointer_cast<BossArm>(m_owner.lock())`**: 持ち主は EnemyBase 型で持っているので、BossArm にしか無いメンバーを触るために変換する。違う型なら空になり `if (!arm)return;` で落ちない。
- **`タイムスケール × m_ownTimeScale` を足していく**: ほかの敵のステート(EnemyAirStay など)と同じ。ヒットストップ・スローで腕も遅くなる。
- **上がるのは EaseOut、戻るのは EaseInOut**: 上がるときは勢いよく飛んでゆっくり止まる(弾き飛ばされた感じ)。戻るときは動き出しと収まりがゆっくり(いきなり速いとワープに見え、最後が速いとくっつく瞬間にガタッとする)。
- **回る速さに `(1 - rate)`**: 上がりきると止まる。ずっと回るとプロペラに見えるし、Step3 で狙いにくい。
- **ふわふわは上がりきってから**: 上昇中に揺れを足すと動きがガタガタになる。
- **戻りの補間先を毎フレーム計算**: ボスは起き上がりのアニメ中で背中が動いているので、切り離した瞬間の形に戻すと最後にワープする。
- **着いたら `ReturnToIdle()`**: 「切り離しの印を消す・戻り具合を0・Idle」を1か所にまとめてある。ImGui の Idle ボタンも同じ関数。
- **Exit が空でも書く**: `EnemyStateBase` の Exit が純粋仮想なので、書かないとビルドできない。

## 2. Step3: 浮いている腕に当たり判定をつける(Step2 が動いてから)

やりたいこと: 打ち上げた腕に、プレイヤーの空中攻撃・スキルを当てたい。そのために、空中で腕を**内部ターゲット(SoftTarget)**として持ち続けたい。

### 調べてわかった今の仕組み(10/27)
- **攻撃が当たる流れ**: `AttackCol` が相手の `HitCol`(やられ判定、`ColRole::Hit` の trigger)に当たる → `HitCol::OnDamageInterFace` → 持ち主の `OnDamage`。
  `HitCol` の位置は毎フレーム「持ち主の `m_rb.m_pos`」(＋offset)。`ColInit` を呼ぶと CollisionManager に自動で登録される。
- **内部ターゲット(SoftTarget)**: `Player::m_softTarget`(weak_ptr)。`GetAttackTarget()` は「ロックオン中ならその敵、違えば内部ターゲット」。
  通常攻撃・スキル・打ち上げ攻撃(PlayerStateAttack / PlayerStateAttackBase / SkillAttack / GuardBreakAttack)はこれを見て向き・吸い寄せを決める。
  - 入る: ① `AttackCol` で最初に当たった敵(ロックオンしていなくて、内部ターゲットがいないとき) ② `CheckNoLockOnTargetEnemy`(入力方向30度以内で一番近い敵)
  - **プレイヤーが空中なら `IsFloor()` が false の敵だけ、地上なら true の敵だけ**を探す(`PlayerStateAttackBase::CheckNoLockOnTargetEnemy`)
  - 消える: 死んだ / ロックオン範囲の3倍より離れた / 攻撃していない時間が90F(`Player::UpdateSoftTarget`)
  - 対象にしない敵は `IsTargetable()` で外している(LockOnManager・PlayerStateAttack の2か所・PlayerStateAttackBase)
- **注意点が3つ見つかった**
  1. `Collider::m_isFloor` の**初期値は true**。腕は CollisionManager に体の当たり判定を登録していないので、ずっと「地上の敵」扱いになる → 空中の対象探しに入らない。
  2. 空中で**入力が無いと、今の内部ターゲットをそのまま使う**(`CheckNoLockOnTargetEnemy` の `if (player->GetSoftTarget())return;`)。
     ブレイクした地上のボスが内部ターゲットのままだと、空中攻撃もボスを狙い続ける。
     (地上版の `PlayerStateAttack` は「地上の内部ターゲットなら使う」と `IsFloor()` を見ているが、空中版には同じ確認が無い)
  3. 内部ターゲットは `IsTargetable()` を見ずに残るので、腕が背中に戻っても内部ターゲットのまま残る。
- **重力・空中の動き**は CollisionManager ではなく各ステートが `m_rb.m_vel` を作っている(`EnemyHitAir` など)。
  腕は `BossArmDetached` が `m_rb.m_pos` を直接決めているので、共通の HitAir などに入れると動きがぶつかる。

### 実装案

**A. 当たり判定の持ち方**
- **案A1(おすすめ)**: やられ判定(`HitCol`)だけ作る。体の当たり判定(押し戻し・物理)は作らない。
  位置は BossArmDetached の軌道のまま。切り離している間だけ `SetIsActive(true)`。一番簡単。
- 案A2: 体の当たり判定も作って物理で動かし、共通の `EnemyHitAir` / `AirStay` を使う。
  打ち上げの見た目を一から作り直すことになり、Step2 の軌道とぶつかるので今はおすすめしない。

**B. 判定の位置(腕の真ん中にする)**
- 今の `m_rb.m_pos` は付け根(肩)。そのままだと、腕の先を叩いても当たらない。
- 案B1(おすすめ): 腕のモデルの肩と手のフレーム(`upperarm_l` と `hand_l` など)の真ん中を `m_rb.m_pos` にする。
  背骨の上書き(`MV1SetFrameUserLocalWorldMatrix`)は今 Draw でしているので、Update でも同じ行列を入れてからフレームの位置を取る。
  形は球(半径は腕の長さの半分くらい)か、肩→手のカプセル。
- 案B2: 切り離した瞬間に「真ん中 − 付け根」のずれを覚えて足す。フレームを読まなくて済むが、回転すると合わない。

**C. 腕が攻撃を受けたとき(`BossArm::OnDamage` を上書き)**
- 敵共通の `OnDamage` はガード・ヒットのステートに移るので、腕では使わない(BossArmDetached が上書きされる)。
- 案C1(おすすめ): ダメージはボスの HP から引く。腕は BossArmDetached のまま、小さく押される・揺れるだけ(`m_detachMove` に足す)。血・ヒットストップは出す。
- 案C2: 腕に別の HP(EnemyStatus.csv に腕の行)。0 になったら腕が壊れて消える流れを作る。

**D. 空中で腕を内部ターゲットに持ち続ける**
1. 切り離している間は `SetIsFloor(false)`(「空中の敵」扱いにする)。戻ったら true に戻す。
2. `IsTargetable()` を「切り離している間(戻り中は除く)だけ true」にする。これでロックオンも腕にできる。
3. `PlayerStateAttackBase::CheckNoLockOnTargetEnemy` の入力なしのとき、**今の内部ターゲットと自分の空中・地上が合うときだけ**使う
   (地上版の PlayerStateAttack と同じ確認)。空中で地上のボスを持っていたら、空中の敵(腕)を探し直す。ボス専用ではなく全部の敵に効く直し方。
4. `Player::UpdateSoftTarget` で `!IsTargetable()` になったら内部ターゲットを外す(腕が背中に戻ったら外れる)。
5. (必要なら)打ち上げ攻撃(GuardBreakAttack の切り上げ)で上がったときに、近い方の腕を内部ターゲットにする。1〜4 で足りなければ。

**E. 空中コンボ中は戻るのを待つ**
- 最後に当たってからのフレームを腕が数え、`RequestReturn` が来ても一定時間(例 60F)当たっていなければ戻る。
  当たっている間は浮いたまま(浮く時間がのびる)。

### 先に決めること
- 腕のダメージはボスの HP か、腕の HP か(案C1 / C2)
- ガードブレイク後の Y の打ち上げはボスに当てたままでよいか(R1 は空中に行かないので浮かない)。腕に当てたいなら、チャンス状態の対象を腕にする必要がある
- 2本の腕のどちらを先に狙うか(入力方向で選ぶ今の仕組みのままでよいか)

## 4. よろめき中に遷移する件(10/27、未解決)

ガードブレイクのよろめき中(`EnemyGuardBreak::CanChangeHit()` が false の間)なのに、攻撃で遷移してしまう。原因は2つ。

1. **ガードブレイクアタックは、わざと判定を飛ばしている**(`BossEnemyPirateR1.cpp` の OnDamage、585行目あたり)
   `if (BossGuardBreak 中 && !isGuardBreakAttack)` の中でしか `CanChangeHit()` を見ていないので、ガードブレイクアタック(Y の打ち上げ、ComboChain01.csv の index 20 / 21、attackType=4)は止まらずに HitGround へ進む。
   コメントに「ガードブレイク攻撃の時はしっかり移行させる」とあるので、打ち上げを必ず当てるための例外。R1 は空中に行かないので、よろめきが切れるだけになっている。
   → 直すなら `&& !isGuardBreakAttack` を外す(使わなくなる `isGuardBreakAttack` の行も消す)。
2. **判定の型を `BossGuardBreak` / `BossGuard` にしたので、普通のガードブレイクを見逃す**(433・447・457・586・588行目あたり)
   デバッグウィンドウの Guard ボタンは普通の `EnemyGuard` を作るので、そこからブレイクすると `EnemyGuardBreak` になり、`dynamic_pointer_cast<BossGuardBreak>` に当てはまらない。
   その場合、どの攻撃でもよろめき中に遷移し、ガード中の受け(`OnGuardHit`)も通らない。
   → **型の判定は親の型(`EnemyGuardBreak` / `EnemyGuard`)で書く**のがおすすめ。BossGuardBreak は EnemyGuardBreak を継承しているので両方に当てはまり、`CanChangeHit()` も親の関数なので呼べる。
